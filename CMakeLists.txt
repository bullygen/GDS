# Сборка самостоятельной C++-библиотеки и необязательного интерфейса Python.
cmake_minimum_required(VERSION 3.18)
project(GDS VERSION 0.1.0 LANGUAGES CXX)
option(GDS_BUILD_PYTHON "Собирать интерфейс Python" OFF)
option(GDS_BUILD_TESTS "Собирать проверки C++" ON)
add_library(gds src/network.cpp src/scheduling.cpp src/solvers.cpp)
add_library(GDS::gds ALIAS gds)
target_compile_features(gds PUBLIC cxx_std_17)
set_target_properties(gds PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(gds PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(gds PRIVATE /W4 /utf-8)
else()
  target_compile_options(gds PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(GDS_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core bindings/module.cpp bindings/network.cpp bindings/scheduling.cpp bindings/solvers.cpp)
  target_link_libraries(_core PRIVATE GDS::gds)
  if(MSVC)
    target_compile_options(_core PRIVATE /utf-8)
  endif()
  install(TARGETS _core LIBRARY DESTINATION gds RUNTIME DESTINATION gds)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS gds EXPORT GDSTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT GDSTargets NAMESPACE GDS:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GDS)
configure_package_config_file(cmake/GDSConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/GDSConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GDS)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/GDSConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GDSConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/GDSConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GDS)
if(GDS_BUILD_TESTS)
  enable_testing()
  add_executable(gds_tests tests/test_core.cpp)
  target_link_libraries(gds_tests PRIVATE GDS::gds)
  add_test(NAME gds_core COMMAND gds_tests)
endif()
